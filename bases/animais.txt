# --- regras ---
# penas só existem em aves (entre os animais atuais)
@regra penas_indicam_ave: (?a tem_propriedade penas) & (?a tem_propriedade poe_ovos) => (?a é_um Ave) 0.95
# amamentar é exclusivo de mamíferos
@regra amamentar_indica_mamifero: (?a tem_propriedade amamenta) => (?a é_um Mamifero) 0.95
# dois predadores da mesma presa disputam alimento
@regra mesma_presa_compete: (?a predador_de ?p) & (?b predador_de ?p) => (?a compete_com ?b) 0.6
# quem tem dieta carnívora e caça outro animal é predador ativo
@regra carnivoro_que_caca: (?a dieta Carnivoro) & (?a predador_de ?p) => (?a tem_propriedade predador_ativo) 0.85
# herança de exceção: o motor herda tem_propriedade da classe, esta regra herda também não_tem_propriedade
@regra heranca_negativa: (?x é_um ?c) & (?c não_tem_propriedade ?p) => (?x não_tem_propriedade ?p) 0.9
# ave aquática: penas e nada
@regra ave_aquatica: (?a tem_propriedade penas) & (?a tem_propriedade nada) => (?a é_um AveAquatica) 0.85
